cmake_minimum_required(VERSION 3.16)
project(access_integrations_tests LANGUAGES C CXX)
enable_testing()
add_executable(access_core_test AccessCoreTest.cpp)
target_compile_features(access_core_test PRIVATE cxx_std_17)
target_include_directories(access_core_test PRIVATE ../include)
if(CJSON_SOURCE)
    target_sources(access_core_test PRIVATE ${CJSON_SOURCE}/cJSON.c)
    target_include_directories(access_core_test PRIVATE ${CJSON_SOURCE})
else()
    find_path(CJSON_INCLUDE cJSON.h PATH_SUFFIXES cjson REQUIRED)
    find_library(CJSON_LIBRARY cjson REQUIRED)
    target_include_directories(access_core_test PRIVATE ${CJSON_INCLUDE})
    target_link_libraries(access_core_test PRIVATE ${CJSON_LIBRARY})
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(access_core_test PRIVATE -Wall -Wextra $<$<COMPILE_LANGUAGE:CXX>:-Werror> -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(access_core_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME access_core_test COMMAND access_core_test)
